@app.get("/webhook")
async def verify_webhook(request: Request):
    mode = request.query_params.get("hub.mode")
    token = request.query_params.get("hub.verify_token")
    challenge = request.query_params.get("hub.challenge", "")
    
    if mode == "subscribe" and token == VERIFY_TOKEN:
        print("✅ Webhook verified successfully!")
        return PlainTextResponse(content=str(challenge), status_code=200)
    
    print("❌ Webhook verification failed: Token mismatch")
    return PlainTextResponse(content="Forbidden", status_code=403)

@app.post("/webhook")
async def handle_messages(request: Request):
    try:
        data = await request.json()
        print(f"📩 Received webhook payload: {data}")
        
        entry = data.get("entry", [])[0]
        changes = entry.get("changes", [])[0]
        value = changes.get("value", {})
        
        if "messages" in value:
            message = value["messages"][0]
            from_number = message["from"]
            msg_body = message.get("text", {}).get("body", "")
            print(f"💬 User Message: '{msg_body}' from {from_number}")
            
            reply_text = f"✈️ Travel Agency Bot: I received your message: '{msg_body}'. How can I help with your booking today?"
            send_whatsapp_message(from_number, reply_text)
            
    except Exception as e:
        print(f"⚠️ Error parsing message payload: {e}")
        
    return {"status": "ok"}

def send_whatsapp_message(to_number: str, text: str):
    url = f"https://graph.facebook.com/v25.0/{PHONE_NUMBER_ID}/messages"
    headers = {
        "Authorization": f"Bearer {ACCESS_TOKEN}",
        "Content-Type": "application/json"
    }
    payload = {
        "messaging_product": "whatsapp",
        "to": to_number,
        "type": "text",
        "text": {"body": text}
    }
    response = requests.post(url, json=payload, headers=headers)
    print(f"📤 Graph API Response: {response.status_code} | {response.text}")
